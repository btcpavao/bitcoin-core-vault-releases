After cloning the repository and installing Docker, run the following commands in Bash. From the release above, put the downloaded ISO next to the repository directory.

```bash
git switch --detach 9f466c5df912d7f455d86aa34d85a39be7b143fc
bash tools/rebuild/build.sh ../vault-result
(
  set -eu
  expected=eb54552a569db6cd1158f712b8713bc6c1b519e17026fa31d4652d4e21157141
  rebuilt=../vault-result/bitcoin-core-vault-amd64.iso
  downloaded=../bitcoin-core-vault-rebuild-9f466c5df912-amd64.iso
  printf '%s  %s\n' "$expected" "$rebuilt" "$expected" "$downloaded" | sha256sum --check
  cmp -- "$rebuilt" "$downloaded"
  echo 'Both ISO files match this record byte for byte.'
)
```
